import assert from 'node:assert/strict';
import {Store} from '../src/store.mjs';
import {homedir} from 'node:os';
import {join} from 'node:path';
const store=new Store(join(homedir(),'.externallink-executor','outbox.sqlite'),{readOnly:true}),jobId='2e7e55c5-4b6d-446f-896a-7491b069f7e3',pair=store.get('pair');let lastCursor;
try{
 for(;;){
  const response=await fetch('http://127.0.0.1:19389/appData',{method:'POST',headers:{Authorization:'Bearer '+pair.localToken,'Content-Type':'application/json'},body:'{}'});assert.equal(response.status,200);const app=await response.json();assert.equal(app.ok,true);const job=app.linkMonitorJobs.find(job=>job.id===jobId);assert.ok(job);assert.equal(job.source,'schedule');
  if(lastCursor!==job.cursor||!job.live){console.log(JSON.stringify({ok:true,kind:'existing_scheduled_monitor_wait',jobId,status:job.status,live:job.live,cursor:job.cursor,count:job.count,counts:job.counts,paused:app.runtime.paused,busy:app.runtime.busy,pendingEdits:app.pendingEdits.length,pendingEvents:app.runtime.pendingEvents}));lastCursor=job.cursor;}
  if(!job.live){assert.ok(['completed','paused'].includes(job.status));break;}
  if(!process.argv.includes('--wait'))break;await new Promise(resolve=>setTimeout(resolve,10000));
 }
}finally{store.close();}
