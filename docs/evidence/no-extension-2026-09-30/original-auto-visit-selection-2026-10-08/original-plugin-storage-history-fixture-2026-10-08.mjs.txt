import assert from 'node:assert/strict';
import {readFile,writeFile} from 'node:fs/promises';
import {join} from 'node:path';
import {Store} from '../src/store.mjs';
import {runHistory} from '../src/run-history.mjs';
const report=JSON.parse(await readFile(new URL('./original-plugin-storage-readonly-2026-10-08.txt',import.meta.url),'utf8')),source=JSON.parse(await readFile(join(report.privateBackupFolder,'private-storage.json'),'utf8')),snapshot=JSON.parse(await readFile(join(report.privateBackupFolder,'snapshot.json'),'utf8'));
assert.equal(report.currentWorkspaceMatches,true);
const history={scope:snapshot.scope,activeBatchRun:source.activeBatchRun,automationRunLedger:source.automationRunLedger},content=JSON.stringify(history),store=new Store(':memory:');
try{
 store.set('pair',{endpoint:source.cloudSyncConfig.endpoint,workspaceId:source.cloudSyncConfig.workspaceId});store.set('paused',true);store.set('acceptanceBatch',{status:'paused',cursor:13,count:30});
 const runtime={store},preview=runHistory(runtime,'previewRunHistory',{content,name:'原插件本机执行历史.json'}),imported=runHistory(runtime,'importRunHistory',{previewId:preview.preview.id}),saved=store.get('originalRunHistory:'+imported.id);
 assert.deepEqual(saved.activeBatchRun,source.activeBatchRun);assert.deepEqual(saved.automationRunLedger,source.automationRunLedger);
 const repeated=runHistory(runtime,'importRunHistory',{previewId:preview.preview.id});assert.equal(repeated.alreadyRestored,true);assert.equal(store.values('originalRunHistory:').length,1);assert.equal(store.values('task:').length,0);assert.equal(store.get('activeBatchRun'),null);assert.equal(store.get('paused'),true);assert.equal(store.get('acceptanceBatch').cursor,13);
 await writeFile(join(report.privateBackupFolder,'original-run-history.json'),content,{flag:'wx'});
 console.log(JSON.stringify({ok:true,kind:'actual_original_plugin_history_import_in_isolated_store',originalCombinations:preview.preview.combinations,originalRuns:preview.preview.runs,originalEvents:preview.preview.events,originalBatchAndLedgerExact:true,repeatedImportDeduplicated:true,noExecutableTaskCreated:true,privateOriginalHistoryFileSaved:true,productionHistoryImported:false,productionBusinessWrites:0,realSubmissions:0}));
}finally{store.close();}
