import assert from 'node:assert/strict';import {readFileSync,writeFileSync,mkdirSync,existsSync} from 'node:fs';import {join} from 'node:path';import {Store} from '../src/store.mjs';import {Cloud} from '../src/cloud.mjs';import {exportApplicationBackup} from '../../core/application-backup.mjs';
const info=JSON.parse(readFileSync(new URL('./migration-reset-ui-fixture-info-2026-10-09.json',import.meta.url),'utf8')),file=join(info.home,'outbox.sqlite'),phase=process.argv[2],store=new Store(file,{readOnly:true}),baselineFile=join(info.home,'ui-baseline.json');
const report={phase,productionWrites:0,models:0,realSubmissions:0};
try{
 if(phase==='before'){
 const pair=store.get('pair'),cloud=new Cloud(pair),snapshot=await cloud.request('snapshot'),baseline={pair,snapshot,task:store.get('task:original-task'),batch:store.get('acceptanceBatch'),pending:store.pending(),learning:store.get('fillLearning:original-learning'),mutation:store.get('appMutation:original-conflict')};writeFileSync(baselineFile,JSON.stringify(baseline));
 const backup=exportApplicationBackup({siteProfiles:{p:{id:'p',name:'原产品',fields:{Name:'原产品',Url:'https://product.example','Long description':'隔离首次迁移补回描述'}}},siteAnnotations:{'migration.example/form':{note:'首次迁移备注',library:{favorite:true,groups:['high_quality']}}},urlList:'https://migration.example/form'});writeFileSync(join(info.home,'ui-import.json'),JSON.stringify(backup));mkdirSync(join(info.home,'browser-profile-stable'),{recursive:true});writeFileSync(join(info.home,'browser-profile-stable','test-marker.txt'),'original-fixture-browser');writeFileSync(join(info.home,'original.png'),'original-fixture-media');report.baselineCaptured=true;
 }else{
 const baseline=JSON.parse(readFileSync(baselineFile,'utf8')),cloud=new Cloud(baseline.pair),snapshot=await cloud.request('snapshot');
 if(phase==='imported'){
 assert.equal(snapshot.documents.siteProfiles.p.fields['Long description'],'隔离首次迁移补回描述');assert.equal(snapshot.documents.siteAnnotations['migration.example/form'].library.favorite,true);assert.deepEqual(snapshot.documents.siteAnnotations['migration.example/form'].library.groups,['high_quality']);assert.deepEqual(store.get('task:original-task'),baseline.task);assert.deepEqual(store.get('acceptanceBatch'),baseline.batch);assert.deepEqual(store.get('fillLearning:original-learning'),baseline.learning);assert.deepEqual(store.get('appMutation:original-conflict'),baseline.mutation);assert.deepEqual(store.pending(),baseline.pending);assert.ok(store.values('backupImport:').some(plan=>plan.status==='completed'));writeFileSync(join(info.home,'ui-after-import.json'),JSON.stringify(snapshot));report.importCloudReadbackAndOriginalStatePreserved=true;
 }else if(phase==='reset'){
 assert.equal(store.get('pair'),null);assert.equal(store.get('task:original-task'),null);assert.equal(store.pendingCount(),0);assert.equal(store.get('paused'),true);assert.deepEqual(snapshot,existsSync(join(info.home,'ui-after-import.json'))?JSON.parse(readFileSync(join(info.home,'ui-after-import.json'),'utf8')):baseline.snapshot);assert.equal(readFileSync(join(info.home,'original.png'),'utf8'),'original-fixture-media');assert.equal(readFileSync(join(info.home,'browser-profile-stable','test-marker.txt'),'utf8'),'original-fixture-browser');
 const backupDirectory=process.argv[3],backup=new Store(join(backupDirectory,'outbox.sqlite'),{readOnly:true});try{assert.deepEqual(backup.get('pair'),baseline.pair);assert.deepEqual(backup.get('task:original-task'),baseline.task);assert.deepEqual(backup.get('acceptanceBatch'),baseline.batch);assert.deepEqual(backup.pending(),baseline.pending);assert.deepEqual(backup.get('appMutation:original-conflict'),baseline.mutation);assert.equal(JSON.parse(readFileSync(join(backupDirectory,'manifest.json'),'utf8')).integrity,'ok');}finally{backup.close();}report.resetBackupIntactCloudAndMediaRetained=true;report.backupDirectory=backupDirectory;
 }
 }
 writeFileSync(new URL('./migration-reset-ui-'+phase+'-2026-10-09.json',import.meta.url),JSON.stringify(report,null,2)+'\n');console.log(JSON.stringify(report));
}finally{store.close();}
